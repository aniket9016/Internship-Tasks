USE taskailogger;

-- Create the GoogleSheetSyncLog table
CREATE TABLE GoogleSheetSyncLog (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    batchNumber BIGINT,
    spreadSheetId VARCHAR(100),
    sheetId VARCHAR(100),
    syncLogStartedOn DATETIME,
    syncLogCompletedOn DATETIME
);

-- Create the GoogleSheetSyncLogErrors table
CREATE TABLE GoogleSheetSyncLogErrors (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    googleSheetSyncLogId BIGINT,
    rowIndex BIGINT,
    columnIndex BIGINT,
    sheetValue VARCHAR(500),
    dbValue VARCHAR(500),
    createdOn DATETIME,
    FOREIGN KEY (googleSheetSyncLogId) REFERENCES GoogleSheetSyncLog(id)
);

USE taskailogger;
DELIMITER //

DROP PROCEDURE IF EXISTS SearchSheetData;

CREATE PROCEDURE SearchSheetData(
    IN batchNumberParam VARCHAR(255),
    IN spreadSheetIdParam VARCHAR(255),
    IN sheetIdParam VARCHAR(255),
    IN searchTextParam VARCHAR(255),  -- Global search
    IN pageNumber INT,                -- Page number (starts from 1)
    IN pageSize INT                   -- Number of records per page
)
BEGIN
    DECLARE offsetValue INT;

    SET offsetValue = (IFNULL(pageNumber, 1) - 1) * IFNULL(pageSize, 10);

    -- Create temporary result set
    CREATE TEMPORARY TABLE IF NOT EXISTS TempResult AS
    SELECT 
        gsl.id AS logId,
        gsl.batchNumber,
        gsl.spreadSheetId,
        gsl.sheetId,
        gse.id AS errorLogId,
        gse.googleSheetSyncLogId,
        gse.rowIndex,
        gse.columnIndex,
        gse.sheetValue,
        gse.dbValue
    FROM 
        GoogleSheetSyncLog gsl
    JOIN 
        GoogleSheetSyncLogErrors gse ON gsl.id = gse.googleSheetSyncLogId
    WHERE 
        (IFNULL(batchNumberParam, '') = '' OR gsl.batchNumber LIKE CONCAT('%', batchNumberParam, '%'))
        AND (IFNULL(spreadSheetIdParam, '') = '' OR gsl.spreadSheetId LIKE CONCAT('%', spreadSheetIdParam, '%'))
        AND (IFNULL(sheetIdParam, '') = '' OR gsl.sheetId LIKE CONCAT('%', sheetIdParam, '%'))
        AND (
            IFNULL(searchTextParam, '') = '' OR (
                gsl.batchNumber LIKE CONCAT('%', searchTextParam, '%') OR
                gsl.spreadSheetId LIKE CONCAT('%', searchTextParam, '%') OR
                gsl.sheetId LIKE CONCAT('%', searchTextParam, '%') OR
                gse.rowIndex LIKE CONCAT('%', searchTextParam, '%') OR
                gse.columnIndex LIKE CONCAT('%', searchTextParam, '%') OR
                gse.sheetValue LIKE CONCAT('%', searchTextParam, '%') OR
                gse.dbValue LIKE CONCAT('%', searchTextParam, '%')
            )
        );

    -- Return paginated results
    IF EXISTS (SELECT 1 FROM TempResult) THEN
        SELECT * FROM TempResult ORDER BY logId LIMIT pageSize OFFSET offsetValue;
    ELSE
        SELECT 'No matching data found.' AS Message;
    END IF;

    DROP TEMPORARY TABLE IF EXISTS TempResult;
END //

DELIMITER ;
CALL SearchSheetData('', '', '', '', 1, 5);
CALL SearchSheetData('1001', '', '', '', 5, 1);
